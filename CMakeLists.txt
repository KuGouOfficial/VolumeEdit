cmake_minimum_required(VERSION 3.20)
project(VolumeEdit VERSION 0.1.0 LANGUAGES CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "VolumeEdit requires Windows")
endif()
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc /guard:cf)
  add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
endif()
enable_testing()
add_library(volume_runtime src/config.cpp src/backup.cpp src/portable.cpp src/audio.cpp)
set(VOLUMEEDIT_DISPLAY_VERSION "v${PROJECT_VERSION_MAJOR}.${PROJECT_VERSION_MINOR}")
if(PROJECT_VERSION_PATCH GREATER 0)
  string(APPEND VOLUMEEDIT_DISPLAY_VERSION ".${PROJECT_VERSION_PATCH}")
endif()
configure_file(src/product_version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/product_version.h @ONLY)
target_include_directories(volume_runtime PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(volume_runtime PUBLIC ole32 uuid propsys shell32 advapi32)
configure_file(src/resources.rc.in ${CMAKE_CURRENT_BINARY_DIR}/resources.rc @ONLY)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/resources.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/volumeedit.ico")
add_executable(VolumeEdit WIN32 src/app.cpp src/app.manifest ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
add_executable(uninstall WIN32 src/uninstall.cpp src/app.manifest ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
foreach(target VolumeEdit uninstall)
  target_link_libraries(${target} PRIVATE volume_runtime comctl32 windowscodecs)
endforeach()
foreach(test config routing backup portable)
  add_executable(${test}_tests tests/${test}_tests.cpp)
  target_link_libraries(${test}_tests PRIVATE volume_runtime)
  add_test(NAME ${test} COMMAND ${test}_tests)
endforeach()
add_test(NAME ui_smoke COMMAND VolumeEdit --smoke)
add_test(NAME uninstall_ui_smoke COMMAND uninstall --smoke)
add_test(NAME self_delete COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_SOURCE_DIR}/tests/self_delete.ps1 -BuildDir ${CMAKE_BINARY_DIR})
add_test(NAME uninstall_cleanup COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_SOURCE_DIR}/tests/uninstall_cleanup.ps1 -BuildDir ${CMAKE_BINARY_DIR})
set_tests_properties(ui_smoke uninstall_ui_smoke self_delete uninstall_cleanup PROPERTIES TIMEOUT 30)
