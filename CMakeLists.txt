cmake_minimum_required(VERSION 3.20)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/Makefile" VOLUMEEDIT_VERSION_LINE
  REGEX "^VERSION[ \t]*=[ \t]*[0-9]+\\.[0-9]+(\\.[0-9]+)?[ \t]*$")
list(LENGTH VOLUMEEDIT_VERSION_LINE VOLUMEEDIT_VERSION_COUNT)
if(NOT VOLUMEEDIT_VERSION_COUNT EQUAL 1)
  message(FATAL_ERROR "Makefile must define one VERSION = major.minor[.patch]")
endif()
string(REGEX REPLACE "^VERSION[ \t]*=[ \t]*([0-9.]+)[ \t]*$" "\\1"
  VOLUMEEDIT_VERSION "${VOLUMEEDIT_VERSION_LINE}")
set(VOLUMEEDIT_DISPLAY_VERSION "v${VOLUMEEDIT_VERSION}")
if(VOLUMEEDIT_VERSION MATCHES "^[0-9]+\\.[0-9]+$")
  string(APPEND VOLUMEEDIT_VERSION ".0")
endif()
project(VolumeEdit VERSION ${VOLUMEEDIT_VERSION} LANGUAGES CXX RC)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/Makefile")
if(NOT WIN32)
  message(FATAL_ERROR "VolumeEdit requires Windows")
endif()
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件: ")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/W4 /utf-8 /permissive- /EHsc /guard:cf)
  add_link_options(/DYNAMICBASE /NXCOMPAT /guard:cf)
endif()
enable_testing()
add_library(volume_runtime src/config.cpp src/backup.cpp src/portable.cpp src/audio.cpp)
configure_file(src/product_version.h.in ${CMAKE_CURRENT_BINARY_DIR}/generated/product_version.h @ONLY)
configure_file(src/app.manifest.in ${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest @ONLY)
configure_file(product.id ${CMAKE_CURRENT_SOURCE_DIR}/out/product.id COPYONLY)
target_include_directories(volume_runtime PUBLIC src ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(volume_runtime PUBLIC ole32 uuid propsys shell32 advapi32)
configure_file(src/resources.rc.in ${CMAKE_CURRENT_BINARY_DIR}/resources.rc @ONLY)
set_source_files_properties(${CMAKE_CURRENT_BINARY_DIR}/resources.rc PROPERTIES OBJECT_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/assets/volumeedit.ico")
add_executable(VolumeEdit WIN32 src/app.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
add_executable(uninstall WIN32 src/uninstall.cpp ${CMAKE_CURRENT_BINARY_DIR}/generated/app.manifest ${CMAKE_CURRENT_BINARY_DIR}/resources.rc)
foreach(target VolumeEdit uninstall)
  target_link_libraries(${target} PRIVATE volume_runtime comctl32 windowscodecs)
  set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY "$<1:${CMAKE_CURRENT_SOURCE_DIR}/out>")
endforeach()
foreach(test config routing backup portable)
  add_executable(${test}_tests tests/${test}_tests.cpp)
  target_link_libraries(${test}_tests PRIVATE volume_runtime)
  add_test(NAME ${test} COMMAND ${test}_tests)
endforeach()
add_test(NAME ui_smoke COMMAND VolumeEdit --smoke)
add_test(NAME uninstall_ui_smoke COMMAND uninstall --smoke)
add_test(NAME self_delete COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_SOURCE_DIR}/tests/self_delete.ps1 -BuildDir ${CMAKE_BINARY_DIR} -ProgramDir $<TARGET_FILE_DIR:uninstall>)
add_test(NAME uninstall_cleanup COMMAND powershell -NoProfile -ExecutionPolicy Bypass -File ${CMAKE_SOURCE_DIR}/tests/uninstall_cleanup.ps1 -BuildDir ${CMAKE_BINARY_DIR} -ProgramDir $<TARGET_FILE_DIR:uninstall>)
set_tests_properties(ui_smoke uninstall_ui_smoke self_delete uninstall_cleanup PROPERTIES TIMEOUT 30)
